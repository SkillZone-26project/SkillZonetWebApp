import React, { useEffect, useState } from "react";
import { useParams, useNavigate } from "react-router-dom";
import {
  MapPin,
  Star,
  Phone,
  Mail,
  Briefcase,
  CheckCircle,
  ArrowLeft,
  MessageCircle,
  CalendarDays,
} from "lucide-react";


const API_BASE_URL =
  "https://skillzonet-backend-auth-v1.onrender.com/api/job/artisans";



const defaultArtisan = {
  id: "default",
  fullName: "John Artisan",
  businessName: "Professional Technician",

  profilePic:
    "https://images.unsplash.com/photo-1500648767791-00dcc994a43e",

  bio:
    "Experienced artisan providing quality services with dedication and professionalism.",

  city: "Nigeria",
  state: "",

  phone: "+234000000000",
  email: "artisan@example.com",

  isAvailable: true,

  averageRating: 4.8,
  totalJobs: 120,
  yearsExperience: 5,

  skills: [],

  reviews: [],
};



const ArtisanProfile = () => {

  const { id } = useParams();

  const navigate = useNavigate();


  const [artisan, setArtisan] = useState(defaultArtisan);

  const [loading, setLoading] = useState(true);

  const [error, setError] = useState("");



  useEffect(() => {

    const fetchArtisanProfile = async () => {

      try {

        setLoading(true);


        if (!id) {

          setArtisan(defaultArtisan);

          return;

        }



        const response = await fetch(

          `${API_BASE_URL}/${id}`

        );



        if (!response.ok) {

          throw new Error(
            "Unable to fetch artisan profile"
          );

        }



        const data = await response.json();



        console.log(
          "Artisan API Response:",
          data
        );



        const artisanData =

          data?.data ||

          data;



        setArtisan({

          ...defaultArtisan,

          ...artisanData,


          profilePic:

            artisanData.profilePic ||

            defaultArtisan.profilePic,



          averageRating:

            artisanData.averageRating || 0,



          totalJobs:

            artisanData.totalJobs || 0,



          yearsExperience:

            artisanData.yearsExperience || 0,



          skills:

            artisanData.skills || [],


          reviews:

            artisanData.reviews || [],


        });



      } catch (err) {


        console.error(
          "Profile fetch error:",
          err
        );


        setError(
          "Unable to load artisan details. Showing default information."
        );


        setArtisan(defaultArtisan);



      } finally {


        setLoading(false);


      }


    };



    fetchArtisanProfile();



  }, [id]);





  const handleBooking = () => {


    navigate(

      `/booking/${artisan.id}`

    );


  };




  const handleMessage = () => {


    navigate(

      `/messages/${artisan.id}`

    );


  };




  if (loading) {


    return (

      <div className="min-h-screen flex items-center justify-center">

        <div className="text-center">


          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600 mx-auto"></div>


          <p className="mt-4 text-gray-600">

            Loading artisan profile...

          </p>


        </div>


      </div>

    );


  }
    return (

    <div className="min-h-screen bg-gray-50 px-4 py-8">


      {error && (

        <div className="max-w-5xl mx-auto mb-5 bg-yellow-100 text-yellow-800 p-3 rounded-lg">

          {error}

        </div>

      )}



      <div className="max-w-5xl mx-auto">



        <button

          onClick={() => navigate(-1)}

          className="flex items-center gap-2 text-gray-700 mb-6 hover:text-blue-600"

        >

          <ArrowLeft size={18} />

          Back

        </button>




        <div className="bg-white rounded-2xl shadow-md overflow-hidden">


          <div className="bg-gradient-to-r from-blue-600 to-indigo-600 h-40"></div>




          <div className="px-6 pb-8">



            <div className="-mt-20 flex flex-col md:flex-row md:items-end gap-5">



              <img

                src={

                  artisan.profilePic ||

                  defaultArtisan.profilePic

                }

                alt={artisan.fullName}

                className="w-36 h-36 rounded-full border-4 border-white object-cover"

              />




              <div className="pb-2">



                <div className="flex items-center gap-2">


                  <h1 className="text-3xl font-bold text-gray-900">


                    {artisan.fullName}


                  </h1>




                  {artisan.isAvailable && (

                    <CheckCircle

                      className="text-blue-600"

                      size={24}

                    />

                  )}


                </div>




                <p className="text-gray-600 mt-1">


                  {artisan.businessName}


                </p>



              </div>



            </div>







            <div className="grid md:grid-cols-3 gap-6 mt-8">



              <div className="flex items-center gap-3 text-gray-700">


                <MapPin

                  className="text-blue-600"

                  size={22}

                />


                <span>

                  {artisan.city}, {artisan.state}

                </span>


              </div>







              <div className="flex items-center gap-3 text-gray-700">


                <Briefcase

                  className="text-blue-600"

                  size={22}

                />


                <span>

                  {artisan.yearsExperience} years experience

                </span>


              </div>







              <div className="flex items-center gap-3 text-gray-700">


                <Star

                  className="text-yellow-500 fill-yellow-500"

                  size={22}

                />


                <span>

                  {artisan.averageRating}

                  ({artisan.reviews?.length || 0} reviews)

                </span>


              </div>



            </div>







            {/* About Artisan */}


            <div className="mt-10">


              <h2 className="text-xl font-bold text-gray-900 mb-3">


                About Artisan


              </h2>



              <p className="text-gray-600 leading-relaxed">


                {artisan.bio || 

                "No biography available for this artisan."}


              </p>


            </div>







            {/* Action Buttons */}


            <div className="flex flex-col sm:flex-row gap-4 mt-8">



              <button

                onClick={handleBooking}

                className="flex items-center justify-center gap-2 bg-blue-600 hover:bg-blue-700 text-white px-6 py-3 rounded-xl transition"

              >

                <CalendarDays size={20}/>

                Book Service


              </button>






              <button

                onClick={handleMessage}

                className="flex items-center justify-center gap-2 border border-blue-600 text-blue-600 hover:bg-blue-50 px-6 py-3 rounded-xl transition"

              >

                <MessageCircle size={20}/>

                Send Message


              </button>



            </div>






            {/* Contact Information */}



            <div className="mt-10 grid md:grid-cols-2 gap-5">



              <div className="bg-gray-50 p-5 rounded-xl">


                <div className="flex items-center gap-3">


                  <Phone

                    className="text-blue-600"

                    size={20}

                  />



                  <div>


                    <p className="text-sm text-gray-500">

                      Phone

                    </p>



                    <p className="font-medium">


                      {artisan.phone || "Not available"}


                    </p>


                  </div>


                </div>


              </div>






              <div className="bg-gray-50 p-5 rounded-xl">



                <div className="flex items-center gap-3">


                  <Mail

                    className="text-blue-600"

                    size={20}

                  />



                  <div>


                    <p className="text-sm text-gray-500">

                      Email

                    </p>



                    <p className="font-medium">


                      {artisan.email || "Not available"}


                    </p>


                  </div>


                </div>


              </div>



            </div>
                        {/* Statistics */}


            <div className="grid grid-cols-2 md:grid-cols-3 gap-5 mt-10">



              <div className="bg-blue-50 rounded-xl p-5 text-center">


                <h3 className="text-2xl font-bold text-blue-600">

                  {artisan.totalJobs}

                </h3>



                <p className="text-gray-600">

                  Completed Jobs

                </p>


              </div>






              <div className="bg-yellow-50 rounded-xl p-5 text-center">


                <h3 className="text-2xl font-bold text-yellow-600">

                  {artisan.averageRating}

                </h3>



                <p className="text-gray-600">

                  Rating

                </p>


              </div>






              <div className="bg-green-50 rounded-xl p-5 text-center">


                <h3 className="text-2xl font-bold text-green-600">

                  {artisan.reviews?.length || 0}

                </h3>



                <p className="text-gray-600">

                  Reviews

                </p>


              </div>



            </div>







            {/* Skills Section */}



            <div className="mt-12">


              <h2 className="text-xl font-bold text-gray-900 mb-4">

                Skills

              </h2>




              <div className="flex flex-wrap gap-3">



                {artisan.skills.map((skill,index)=>(


                  <span

                    key={index}

                    className="bg-blue-100 text-blue-700 px-4 py-2 rounded-full text-sm font-medium"

                  >


                    {

                      skill.subSkill?.name ||

                      skill.subSkill?.skill?.name ||

                      "Skill"

                    }


                  </span>


                ))}



              </div>


            </div>







            {/* Services Section */}



            <div className="mt-12">


              <h2 className="text-xl font-bold text-gray-900 mb-5">

                Services Offered

              </h2>




              <div className="grid md:grid-cols-2 gap-6">



                {artisan.skills.map((service,index)=>(



                  <div

                    key={index}

                    className="border rounded-xl p-5 hover:shadow-md transition"

                  >



                    <h3 className="font-bold text-lg text-gray-900">


                      {

                        service.subSkill?.skill?.name ||

                        "Professional Service"

                      }


                    </h3>





                    <p className="text-gray-600 mt-2">


                      {

                        service.subSkill?.name ||

                        "Quality service delivered professionally."

                      }


                    </p>





                    <div className="mt-4 text-blue-600 font-semibold">


                      Contact for pricing


                    </div>




                  </div>


                ))}



              </div>


            </div>







            {/* Reviews Section */}



            <div className="mt-12">


              <h2 className="text-xl font-bold text-gray-900 mb-5">

                Customer Reviews

              </h2>




              <div className="bg-gray-50 rounded-xl p-6">



                {artisan.reviews?.length > 0 ? (



                  artisan.reviews.map((review,index)=>(


                    <div

                      key={index}

                      className="bg-white p-4 rounded-lg mb-4"

                    >


                      <p className="font-semibold">

                        Customer Review

                      </p>


                      <p className="text-gray-600 text-sm">

                        {review.comment || "Great service"}

                      </p>



                    </div>


                  ))



                ) : (



                  <p className="text-gray-600">

                    No reviews available yet.

                  </p>


                )}



              </div>



            </div>








            {/* Availability Section */}



            <div className="mt-12">



              <h2 className="text-xl font-bold text-gray-900 mb-4">

                Availability

              </h2>





              <div className="bg-green-50 rounded-xl p-5 flex items-center gap-3">



                <CheckCircle

                  className="text-green-600"

                  size={24}

                />




                <p className="text-green-700 font-medium">


                  {artisan.isAvailable

                    ? "Artisan is available for new jobs"

                    : "Artisan is currently unavailable"

                  }



                </p>



              </div>



            </div>








            {/* More Information */}



            <div className="mt-12">



              <h2 className="text-xl font-bold text-gray-900 mb-4">

                More Information

              </h2>





              <p className="text-gray-600">


                You can contact this artisan directly to discuss your project requirements, pricing, and availability.


              </p>



            </div>





          </div>



        </div>



      </div>



    </div>


  );


};



export default ArtisanProfile;
