import React, { useEffect, useRef, useState } from 'react'

import {
  LuCamera,
  LuPhone,
  LuCalendar,
  LuSquarePen
} from "react-icons/lu"; 
               
import { HiOutlineEnvelope } from "react-icons/hi2";
import { CiLocationOn } from "react-icons/ci";
import axios from "axios";

const UserProfile = () => {

  const defaultUser = {
    fullName: "Sarah Johnson",
    role: "Client",
    email: "sarah.johnson@email.com",
    city: "Lekki",
    state: "Lagos",
    country: "Nigeria",
    phone: "+234 803 456 7890",
    profilePic:
      "https://res.cloudinary.com/dqtyrjpeh/image/upload/v1774203306/Primitive.span_2_bssqdu.png",
    createdAt: null
  };

  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const fileInputRef = useRef(null);

  useEffect(() => {
    const fetchProfile = async () => {
      try {
        setLoading(true);

        const token = localStorage.getItem("token");

        if (!token) {
          setUser(defaultUser);
          return;
        }

        const res = await axios.get(
          "https://skillzonet-backend-auth-v1.onrender.com/api/userAuth/get-profile",
          {
            headers: {
              Authorization: `Bearer ${token}`
            }
          }
        );

        const apiUser = res.data?.user ?? res.data ?? {};

        setUser({
          fullName: apiUser?.fullName || defaultUser.fullName,
          role: apiUser?.role || defaultUser.role,
          email: apiUser?.email || defaultUser.email,
          city: apiUser?.city || defaultUser.city,
          state: apiUser?.state || defaultUser.state,
          country: apiUser?.country || defaultUser.country,
          phone: apiUser?.phone || defaultUser.phone,
          profilePic: apiUser?.profilePic || defaultUser.profilePic,
          createdAt: apiUser?.createdAt || null
        });

      } catch (err) {
        setUser(defaultUser);
      } finally {
        setLoading(false);
      }
    };

    fetchProfile();
  }, []);

  const handleCameraClick = () => {
    fileInputRef.current?.click();
  };

  const joinedDate = user?.createdAt ? new Date(user.createdAt) : null;

  const joinedMonth = joinedDate && !isNaN(joinedDate)
    ? joinedDate.toLocaleString("default", { month: "long" })
    : null;

  const joinedYear = joinedDate && !isNaN(joinedDate)
    ? joinedDate.getFullYear()
    : null;

  const handleFileChange = async (e) => {
    const file = e.target.files?.[0];
    if (!file) return;

    try {
      const token = localStorage.getItem("token");

      const formData = new FormData();
      formData.append("file", file);

      const res = await axios.post(
        "https://skillzonet-backend-auth-v1.onrender.com/api/userAuth/profile-picture",
        formData,
        {
          headers: {
            Authorization: `Bearer ${token}`,
            "Content-Type": "multipart/form-data"
          }
        }
      );

      const newImage =
        res.data?.profilePic || res.data?.user?.profilePic;

      if (newImage) {
        setUser(prev => ({
          ...prev,
          profilePic: newImage
        }));
      }

    } catch (err) {
      console.log(err);
    }
  };

  return (
    <div className="h-[calc(100vh-85px)] bg-white mt-[85px] px-4 rounded-[20px] p-[24px] ml-[100px]">

      {/* TOP SECTION */}
      <div className='left and middle border h-[361px] p-[24px] rounded-[14px] w-[855px]'>

        <div className="topBox flex gap-[50px] h-[210px] border-b">

          <div className='left flex gap-[15px]'>

            <div className="relative w-[128px] h-[128px]">

              {/* IMAGE FIX (NO FLASH) */}
              {loading ? (
                <div className="w-[128px] h-[128px] rounded-full bg-gray-200 animate-pulse" />
              ) : (
                <img
                  src={user?.profilePic}
                  className="w-[128px] h-[128px] rounded-[64px]"
                  alt=""
                />
              )}

              <div
                onClick={handleCameraClick}
                className="absolute bottom-0 right-0 bg-black text-white w-[40px] h-[40px] rounded-[20px] flex items-center justify-center cursor-pointer"
              >
                <LuCamera />
              </div>

              <input
                type="file"
                ref={fileInputRef}
                accept="image/*"
                style={{ display: "none" }}
                onChange={handleFileChange}
              />

            </div>

            <div>
              <p className='text-[24px] font-semibold text-textColor mt-[5px]'>
                {user?.fullName}
              </p>

              <p className='text-[14px] font-normal text-textGray mt-[5px]'>
                {user?.role}
              </p>

              <div className='flex items-center gap-[10px] mt-[10px]'>
                <HiOutlineEnvelope />
                <p>{user?.email}</p>
              </div>

              {/* <div className='flex items-center gap-[10px] mt-[10px]'>
                <CiLocationOn />
                <p>{user?.city}, {user?.state}, {user?.country}</p>
              </div>
                 */}
            </div>

          </div>

          <div className="rightTop flex justify-between w-[350px]">

            <div className="flex flex-col justify-end">

              <div className='flex items-center'>
                <LuPhone />
                <p className='text-[14px]'>{user?.phone}</p>
              </div>

              <div>
                <div className='flex items-center gap-[10px] mt-[15px]'>
                  <LuCalendar />
                  <p>Joined {joinedMonth || "June"}</p>
                </div>

                <p className='ml-[30px]'>
                  {joinedYear || "2024"}
                </p>
              </div>

            </div>

            <button className='flex items-center justify-center gap-[10px] w-[127px] h-[36px] rounded-[8px] bg-black text-white'>
              <LuSquarePen />
              <p>Edit Profile</p>
            </button>

          </div>

        </div>

        {/* SECOND SECTION */}
        <div className="secondDown flex gap-[16px] mt-[50px]">

          <div className="box1 w-[187px] flex flex-col items-center">
            <p className='text-[24px] font-semibold text-textColor'>12</p>
            <p className='text-[14px] font-normal text-textGray'>Total Bookings</p>
          </div>

          <div className="box2 w-[187px] flex flex-col items-center">
            <p className='text-[24px] font-semibold text-textColor'>5</p>
            <p className='text-[14px] font-normal text-textGray'>Saved Artisans</p>
          </div>

          <div className="box1 w-[187px] flex flex-col items-center">
            <p className='text-[24px] font-semibold text-textColor'>₦<span>245,000</span></p>
            <p className='text-[14px] font-normal text-textGray'>Total Spent</p>
          </div>

          <div className="box1 w-[187px] flex flex-col items-center">
            <p className='text-[24px] font-semibold text-textColor'>8</p>
            <p className='text-[14px] font-normal text-textGray'>Reviews Given</p>
          </div>

        </div>

      </div>

      {/* PERSONAL INFO SECTION */}
      <div className="middleBox w-[855px] border px-[24px] py-[26px] mt-[24px] rounded-[14px]">

        <div className='flex justify-between'>
          <p>Personal Information</p>

          <button className='flex items-center justify-center gap-[10px] w-[77px] h-[36px] rounded-[8px] text-[14px] text-textColor border'>
            <LuSquarePen />
            <p>Edit</p>
          </button>
        </div>

        <div>
          <form>
            <fieldset className='flex gap-[10px]'>

              <div className='flex flex-col'>
                <label>Full Name</label>
                <input className='w-[391px] bg-[#F3F3F5] h-[36px] rounded-[8px] pl-[12px]' />
              </div>

              <div className='flex flex-col'>
                <label>Email Address</label>
                <input className='w-[391px] bg-[#F3F3F5] h-[36px] rounded-[8px] pl-[12px]' />
              </div>

            </fieldset>

            <fieldset className='flex gap-[10px]'>

              <div className='flex flex-col'>
                <label>Phone Number</label>
                <input className='w-[391px] bg-[#F3F3F5] h-[36px] rounded-[8px] pl-[12px]' />
              </div>

              <div className='flex flex-col'>
                <label>Location</label>
                <input className='w-[391px] bg-[#F3F3F5] h-[36px] rounded-[8px] pl-[12px]' />
              </div>

            </fieldset>

          </form>
        </div>

      </div>

    </div>
  );
};

export default UserProfile;
